void oled_show_string(int page, int col, const char *text) {
    if (page<0||page>7) return; if (col<0||col>122) col=0;
    oled_write_cmd(0xB0+page); oled_write_cmd(col&0x0F); oled_write_cmd(0x10|(col>>4));
    while (*text) {
        if (*text<32||*text>126) { text++; continue; }
        const uint8_t *glyph = font6x8[*text-32];
        for (int i=0;i<6;i++) oled_write_data(glyph[i]);
        text++;
    }
}
void oled_show_nav(const NavData *nav) {
    oled_write_cmd(0xB0);oled_write_cmd(0x00);oled_write_cmd(0x10);for(int c=0;c<128;c++)oled_write_data(0x00);oled_write_cmd(0xB2);oled_write_cmd(0x00);oled_write_cmd(0x10);for(int c=0;c<128;c++)oled_write_data(0x00);
    char line1[32], line2[32];
    snprintf(line1, sizeof(line1), "Dir: %s", nav->direction_text);
    snprintf(line2, sizeof(line2), "Dist: %dm", nav->distance);
    oled_show_string(0, 0, line1);
    oled_show_string(2, 0, line2);
}
